1 def delete(node, val):
2     # Base case: value not found in tree
3     if node is None:
4         return None
5
6     # Search for the node to delete
7     if val < node.val:
8         node.left = delete(node.left, val)
9     elif val > node.val:
10         node.right = delete(node.right, val)
11
12     else: # Node to delete found
13         if node.left is None and node.right is None:
14             return None # Leaf node: just remove it
15
16         if node.left is None:
17             return node.right # One child on right
18         if node.right is None:
19             return node.left # One child on left
20
21         # Two children: find inorder successor
22         succ = node.right
23         while succ.left:
24             succ = succ.left
25         node.val = succ.val # Replace with successor
26         node.right = delete(node.right, succ.val)
27
28     return node
