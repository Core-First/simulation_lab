1 def inorder(node, result=[]):
2     # In-order: Left → Root → Right (produces sorted output)
3     if node:
4         inorder(node.left, result)
5         result.append(node.val)
6         inorder(node.right, result)
7     return result
8
9 def preorder(node, result=[]):
10     # Pre-order: Root → Left → Right (good for copying)
11     if node:
12         result.append(node.val)
13         preorder(node.left, result)
14         preorder(node.right, result)
15     return result
16
17 def postorder(node, result=[]):
18     # Post-order: Left → Right → Root (good for deletion)
19     if node:
20         postorder(node.left, result)
21         postorder(node.right, result)
22         result.append(node.val)
23     return result
